Tutorial 1: Building a Simple Duel

This tutorial walks you from zero to a functioning two-player turn loop: draw, play a creature, attack, and check for a winner.

Step 1 — Create data assets

In your Unity project, create the following ScriptableObject assets (all under Create → Card Game Engine):

  • ZoneDefinition — Deck (DefaultInsertPosition: Bottom, IsOrdered: true)
  • ZoneDefinition — Hand (DefaultInsertPosition: Top, MaxCapacity: 7)
  • ZoneDefinition — Battlefield (DefaultInsertPosition: Top, IsOrdered: false)
  • ZoneDefinition — Graveyard (DefaultInsertPosition: Top)
  • StatDefinition — Power
  • StatDefinition — Toughness
  • CardType — Creature
  • CardDefinition — Goblin Raider (Types: Creature, Stats: Power=2, Toughness=1)
  • CardTemplate — assign Goblin Raider as the face

Step 2 — Bootstrap the game

C# — DuelBootstrap.cs 
using System.Linq;
using UnityEngine;
using CardGameEngine;

public class DuelBootstrap : MonoBehaviour
{
    public ZoneDefinition deckDef, handDef, battlefieldDef, graveyardDef;
    public CardTemplate   goblinTemplate;
    public StatDefinition  powerDef, toughnessDef;

    private GameContext _ctx;
    private Player      _p1, _p2;
    private Zone        _p1Deck, _p1Hand, _battlefield, _graveyard;

    void Start()
    {
        _ctx = new GameContext();
        _p1  = new Player { DisplayName = "Alice", Life = 20 };
        _p2  = new Player { DisplayName = "Bob",   Life = 20 };

        _p1Deck     = new Zone(deckDef,        _p1);
        _p1Hand     = new Zone(handDef,        _p1);
        _battlefield= new Zone(battlefieldDef, null);
        _graveyard  = new Zone(graveyardDef,   null);

        _ctx.ZoneManager.Register(_p1Deck);
        _ctx.ZoneManager.Register(_p1Hand);
        _ctx.ZoneManager.Register(_battlefield);
        _ctx.ZoneManager.Register(_graveyard);

        // Fill the deck with 20 Goblin Raiders
        for (int i = 0; i < 20; i++)
            _p1Deck.TryAdd(new CardInstance(goblinTemplate, _p1, _p1));

        // Shuffle and draw an opening hand of 7
        _p1Deck.Shuffle();
        DrawCards(_p1, _p1Deck, _p1Hand, 7);

        Debug.Log($"Hand size: {_p1Hand.Count}");
        Debug.Log($"Deck remaining: {_p1Deck.Count}");
    }

    void DrawCards(Player player, Zone deck, Zone hand, int count)
    {
        var source = new EffectSource("Draw");
        new DrawEffect(source, player, hand, count).Execute(_ctx);
    }
}

Step 3 — Play a creature and attack

C# — Playing and attacking 
void PlayCreatureFromHand(CardInstance card)
{
    var source = new EffectSource(card);
    new MoveEffect(source, _p1, card, _battlefield).Execute(_ctx);
    Debug.Log($"{card.Face.CardName} entered the battlefield!");
}

void AttackWithCreature(CardInstance attacker, Player defender)
{
    var source = new EffectSource(attacker);
    var state  = _ctx.ContinuousEffectManager.ComputeState(attacker);
    int power  = state.GetStat(powerDef);

    new DamageEffect(source, _p1, defender, power).Execute(_ctx);
    Debug.Log($"{attacker.Face.CardName} attacks for {power}! {defender.DisplayName} is at {defender.Life} life.");
}

void CheckWinCondition()
{
    if (_p2.Life <= 0)
        Debug.Log($"{_p1.DisplayName} wins!");
}

Tutorial 2: Writing a Custom Spell

Let's implement "Lava Spike" — deal 3 damage to any target.

C# — LavaSpikeAbility.cs 
using CardGameEngine;

/// <summary>Lava Spike — Sorcery. Deal 3 damage to any target.</summary>
public sealed class LavaSpikeEffect : EffectBase
{
    private readonly ITargetable _target;

    public LavaSpikeEffect(EffectSource src, Player ctrl, ITargetable target)
        : base(src, ctrl)
    {
        _target = target;
    }

    public override EffectResult Execute(GameContext ctx)
    {
        if (_target == null) return EffectResult.Fizzled;

        return new DamageEffect(Source, Controller, _target, 3)
            .Execute(ctx);
    }
}

Tutorial 3: Triggered Abilities

Let's implement "Whenever a creature enters the battlefield, draw a card." — a classic ETB trigger.

C# — ETB Draw trigger 
// Register the trigger once during game setup
context.EventBus.Subscribe<CardEnteredZoneEvent>(e =>
{
    // Only trigger when a creature enters the battlefield
    if (e.Zone != battlefield) return;
    if (!e.Card.IsType(creatureType)) return;

    // Queue the draw effect (execute at end of the current phase in a real game)
    var source = new EffectSource("ETB Draw Trigger");
    new DrawEffect(source, e.Card.Controller, hand, 1).Execute(context);

    Debug.Log($"Trigger: drew a card because {e.Card.Face.CardName} entered.");
});

Tutorial 4: +1/+1 Counter Buff

A common pattern in deck-builders: give a creature a +1/+1 counter. The counter's stat effect is configured on the CounterType asset.

C# — Adding a +1/+1 counter 
// counters.plusOnePlusOne is a CounterType SO with HasStatEffect=true,
// ValuePerCounter=1 for both Power and Toughness.

// Add 3 counters to a creature
creature.AddCounters(counters.plusOnePlusOne, 3);

// Read the buffed stats
var state     = context.ContinuousEffectManager.ComputeState(creature);
int power     = state.GetStat(powerDef);     // base + 3
int toughness = state.GetStat(toughnessDef); // base + 3
Debug.Log($"{creature.Face.CardName} is now {power}/{toughness}");
